\section*{Extended Executive Summary of Contributions for \textit{JMIS}}
\begin{enumerate}[leftmargin=1.8em, itemsep=6pt]
  \item \textbf{Dual-Cue Epistemic Defeasibility Model (DCEDM, Theorem 1 \& Corollary 1):} Integrates metacognitive processing fluency theory, human-AI delegation, Toulmin defeasible argumentation, and endogenous neuro-cognitive warning habituation into a normative Bayesian decision-theoretic governance model for SLA-constrained SOC alert triage.
  \item \textbf{Prompt-Isolated Telemetry Discrepancy Scorer (PITSA) \& Two-Stage DECS (Proposition 1, DP1 to DP3):} Enforces a mathematical air-gap ($\Pi_{\text{struct}}$, $\nabla_{\mathcal{A}_{\text{text}}} L(e) \equiv 0$) between kernel/flow structural pillars ($\mathcal{A}_{\text{struct}}$) and attacker-controllable strings ($\mathcal{A}_{\text{text}}$), extracting exact per-pillar forensic defeaters ($k^*(e)$) in $2.12$ to $4.31\text{ }\mu\text{s}$ per event on a single CPU core.
  \item \textbf{Multi-Corpus Zero-Leakage Empirical Validation ($N = 2,238,434$ Records, $671,531$ Held-Out Test Records):} Across \texttt{OTRF Security-Datasets}, \texttt{SALAD-SOC}, \texttt{LMD-2023}, and \texttt{CISA KEV}, five unpartitioned lexical, dense semantic neural MLP, and subword language model classifiers suffer statistically significant recall collapse under graded appended ($m \in \{1,2,4,8\}$) and full-override ITPI (McNemar's $\chi^2 \ge 86.01$ on OTRF and $\chi^2 \ge 8,361.46$ on SALAD-SOC, $p < 0.0001$), whereas Partitioned PITSA maintains $0.00\text{ pp}$ degradation ($\text{AUROC} = 0.9807$ on OTRF, $0.9380$ on SALAD-SOC, and $0.9963$ on LMD-2023).
  \item \textbf{Trace-Driven Governance Dominance Boundaries and Habituation Optimization:} Selective DECS ($\Pi_4$) reduces Compromise Miss Rates by $18.45\text{ pp}$ (OTRF) and $37.32\text{ pp}$ (SALAD-SOC) at $\lambda_F = 0.40$, reduces expected operational loss by $30.16\%$ to $60.77\%$ on SALAD-SOC ($\kappa^* = 1.68$), and achieves an interior U-shaped optimal gating threshold at $\tau^*_{\text{LLR}} = 1.0$.
\end{enumerate}
